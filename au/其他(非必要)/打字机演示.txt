extends Node2D # 或者 Control、Node，看你挂在什么节点上

func _ready():
	# 注意：var 以及下面的内容，前面都必须有一个 Tab 的缩进
	var 完整配置 = {
		# --- 打字机核心 ---
		"打字间隔": 0.05,        # 每个字间隔（秒），越小越快
		
		# --- 字体与颜色 ---
		"字体路径": "res://fonts/MonsterFriendFore.otf",  # 填你项目里的字体路径
		"字体大小": 20,          # 字号
		"默认颜色": Color(1, 1, 1, 1),                    # 默认颜色（白色）
		"自定义颜色字符串": "",                            # 形如 "255,0,0.0,255,0"，留空 = 单色
		
		# --- 音效 ---
		"音效路径": "res://sounds/snd_wngdng7.wav",       # 填你项目里的音效路径
		"音效音量": 1.0,         # 0.0 到 1.0
		
		# --- 位置 ---
		# 不填X和Y，或者都填0，会自动在屏幕正中间显示
		"X": 0,                  # 距离屏幕左边的像素
		"Y": 0,                  # 距离屏幕顶部的像素
		
		# --- 首字符（比如说话人名字前缀）---
		"首字符": "*",           # 显示在正文前面的字符
		"首字符静音": true,       # 首字符打字时是否静音
		
		# --- 按键推进 ---
		"启用按键推进": true,      # 是否允许按键推进/跳过
		"下一个动作名": &"confirm", # 打字中按 = 瞬间显示；打完后按 = 请求下一条
		"跳过动作名": &"cancel",   # 按 = 立即完成当前句
		
		# --- 文字背景框 ---
		"启用文字背景框": true,
		"背景框颜色": Color.GREEN,
		"背景框外扩": 0.0,        # 背景框比文字大多少像素
		
		# --- 外框 ---
		"启用外框": true,
		"外框颜色": Color.WHITE,
		"外框宽度": 6.0,
	}
	
	# 调用全局单例
	打字机.开始打印("Hello World", 完整配置)
