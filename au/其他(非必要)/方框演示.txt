extends Node2D # 或者 Control、Node，看你挂在什么节点上

func _ready():
	# 注意：var 以及下面的内容，前面都必须有一个 Tab 的缩进
	var 完整配置 = {
		# --- 角色设置 ---
		"角色贴图": preload("res://icon.svg"),   # 填 Texture2D 资源路径，比如 preload("res://icon.svg")；留空 = null 就只显示方框不显示角色
		"角色移动速度": 200.0,                    # 填数字（像素/秒），比如 200 = 每秒走 200 像素；值越大走得越快
		"角色初始偏移": Vector2.ZERO,             # 填 Vector2(x, y)，比如 Vector2.ZERO = 正好在方框正中央；Vector2(50, 0) = 往右偏 50 像素
		
		# --- 方框设置 · 单位 = 像素格 ---
		"方框尺寸": Vector2i(138, 46),            # 填 Vector2i(宽, 高)，单位是像素格，比如 Vector2i(138, 46) = 宽 138 格、高 46 格
		"边框厚度": 2,                            # 填整数（像素格），比如 2 = 边框厚 2 格；值越大边框越粗
		"圆角半径": 4,                            # 填整数（像素格），比如 4 = 四个角圆润 4 格；0 = 直角，值越大角越圆
		"像素格大小": 4.0,                        # 填小数，比如 4.0 = 一个像素格占屏幕 4×4 像素；值越大方框整体越大
		"居中显示": true,                          # 填 true 或 false，true = 方框以节点原点为中心；false = 从节点原点向右下画
		
		# --- 方框形状 ---
		"形状类型": 0,                            # 填 0 或 1，0 = 矩形，1 = 圆形（代表方框的初始形状）
		"圆形半径": -1.0,                         # 填小数（像素格），比如 20.0 = 圆半径 20 格；填 -1.0 = 自动用方框短边的一半（内切圆）
		"变形时间": 0.5,                          # 填小数（秒），比如 0.5 = 变形动画用 0.5 秒完成；调用时不传时间就用这个默认值
		
		# --- 渐变流动 ---
		"渐变": null,                             # 填 Gradient 资源，比如 load("res://颜色/彩虹.tres")；留空 = null 就自动用彩虹色循环
		"渐变周期数": 1.0,                        # 填小数，比如 1.0 = 方框横向铺满一条完整渐变；2.0 = 铺两条（条纹更密）
		"流动速度": 0.3,                          # 填小数，比如 0.3 = 每秒移动 0.3 个渐变周期；正数向右流，负数向左流，0 = 不流动
	}
	
	# 找到你的主角节点，把配置一次性应用上去
	var 主角 = $主角  # 改成你场景里实际的主角节点路径，比如 $Player 或 get_node("主角")
	主角.应用配置(完整配置)
